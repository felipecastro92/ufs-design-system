Trade Partner Selector

Webshop band under the breadcrumb: pick a city and a trade partner (searchable dropdowns), then see the selection with Change and Connect. Connect opens a drawer about ordering through the partner's platform.

Source
handover/v2/
template-products-catalog-webshop.html
Format·HTML + CSS + JS
Overview

Full Component Preview

The Trade Partner Selector is a band directly under the breadcrumb on webshop pages (Products catalog, Product detail, Checkout). The chef picks a city, then a trade partner in that city. The band then shows the selection with Change and, when the partner has an online ordering platform, Connect with trade partner, which opens a drawer explaining how ordering through that platform works. A collapsible note under the controls explains the service.

💡
Live: pick Johannesburg to see Connect (only its partners have a platform in the example data). Any other city shows the selection without Connect.

      Structure

      Anatomy

      One root, .ufs-tps, with two steps switched by data-state, a visually hidden status line, and the note. The dropdowns are a .fi-field trigger button plus a popover with a search field and a listbox (the same pattern as the sign-up form's country-code dropdown). They aren't native <select>s, because the lists need search.

      The selection is session-only in the prototype; persist it per account (or per session for guests) so the band opens in the Selected step on the next page.

var tps = { city: '', selection: null, open: null, active: -1, trigger: null };
function tpsEl(id) { return document.getElementById(id); }

/* --- Dropdowns (city / partner) --- */
function tpsDropOptions(key) {
  return key === 'city' ? Object.keys(TPS_PARTNERS)
    : (TPS_PARTNERS[tps.city] || []).map(function (p) { return p.name; });
}
function tpsMatches(key, query) {
  var q = (query || '').trim().toLowerCase();
  return tpsDropOptions(key).filter(function (o) { return o.toLowerCase().indexOf(q) !== -1; });
}
function tpsDropSetValue(key, value) {
  var trigger = tpsEl('tps-' + key), text = value || TPS_TEXT[key].placeholder;
  trigger.textContent = text;
  trigger.setAttribute('aria-label', TPS_TEXT[key].name + ': ' + text);
  trigger.dataset.value = value || '';
}
function tpsDropRender(key, query) {
  var current = tpsEl('tps-' + key).dataset.value, ul = tpsEl('tps-' + key + '-list');
  var matches = tpsMatches(key, query);
  ul.innerHTML = '';
  tpsSetActive(key, -1);
  if (!matches.length) { ul.innerHTML = '<li class="ufs-tps__empty" role="presentation">' + TPS_TEXT.noMatches + '</li>'; return; }
  matches.forEach(function (o, i) {
    var li = document.createElement('li');
    li.className = 'ufs-tps__item';
    li.id = 'tps-' + key + '-opt-' + i;
    li.setAttribute('role', 'option');
    li.setAttribute('aria-selected', o === current ? 'true' : 'false');
    li.textContent = o;
    // mousedown, not click: keeps focus in the search field so the popover doesn't blur-close first
    li.addEventListener('mousedown', function (e) { e.preventDefault(); tpsDropSelect(key, o); });
    ul.appendChild(li);
  });
}
// Keyboard highlight: ↑ / ↓ move it, the search field keeps focus (aria-activedescendant)
function tpsSetActive(key, i) {
  var items = tpsEl('tps-' + key + '-list').querySelectorAll('.ufs-tps__item');
  tps.active = i;
  items.forEach(function (li, n) { li.classList.toggle('is-active', n === i); });
  var search = tpsEl('tps-' + key + '-search');
  if (i >= 0 && items[i]) { search.setAttribute('aria-activedescendant', items[i].id); items[i].scrollIntoView({ block: 'nearest' }); }
  else search.removeAttribute('aria-activedescendant');
}
function tpsDropClose() {
  document.querySelectorAll('.ufs-tps__panel').forEach(function (p) { p.hidden = true; });
  document.querySelectorAll('.ufs-tps__select').forEach(function (t) { t.setAttribute('aria-expanded', 'false'); });
  document.querySelectorAll('.ufs-tps__field').forEach(function (f) { f.classList.remove('is-open'); });
  tpsEl('trade-partner-selector').classList.remove('is-drop-open');
  tps.open = null;
}
function tpsDropToggle(key) {
  var wasOpen = tps.open === key;
  tpsDropClose();
  if (wasOpen) return;
  tpsEl('tps-' + key + '-panel').hidden = false;
  tpsEl('tps-' + key).setAttribute('aria-expanded', 'true');
  tpsEl('tps-' + key).parentNode.classList.add('is-open');
  tpsEl('trade-partner-selector').classList.add('is-drop-open');
  tps.open = key;
  var search = tpsEl('tps-' + key + '-search');
  search.value = '';
  tpsDropRender(key, '');
  search.focus();
}
function tpsDropSelect(key, value) {
  tpsDropClose();
  if (key === 'city') { tpsEl('tps-city').focus(); tpsSelectCity(value); }
  else tpsSelectPartner(value);
}
function tpsDropKeydown(e, key) {
  var n = tpsMatches(key, e.target.value).length;
  if (e.key === 'Escape') { e.preventDefault(); e.stopPropagation(); tpsDropClose(); tpsEl('tps-' + key).focus(); return; }
  if (e.key === 'ArrowDown' || e.key === 'ArrowUp') {
    e.preventDefault();
    if (n) tpsSetActive(key, e.key === 'ArrowDown' ? (tps.active + 1) % n : (tps.active <= 0 ? n - 1 : tps.active - 1));
    return;
  }
  if (e.key === 'Tab') { tpsDropClose(); return; }
  if (e.key !== 'Enter') return;
  e.preventDefault();
  var match = tpsMatches(key, e.target.value)[Math.max(0, tps.active)];   // highlighted, else the first match
  if (match) tpsDropSelect(key, match);
}
// pointerdown, not click: iOS Safari doesn't deliver taps on non-interactive areas to a click listener
document.addEventListener('pointerdown', function (e) {
  if (tps.open && !e.target.closest('.ufs-tps__field')) tpsDropClose();
});

/* --- Steps --- */
function tpsSelectCity(city) {
  if (city === tps.city) return;
  tps.city = city;
  tpsDropSetValue('city', city);
  tpsDropSetValue('partner', '');
  tpsEl('tps-partner').disabled = !city;
  tpsEl('trade-partner-selector').classList.toggle('has-city', !!city);
}
function tpsSelectPartner(name) {
  var city = tps.city;
  var partner = (TPS_PARTNERS[city] || []).filter(function (p) { return p.name === name; })[0];
  if (!partner) return;
  var root = tpsEl('trade-partner-selector'), triggers = root.querySelectorAll('.ufs-tps__select');
  var summary = partner.name + ' - ' + city;
  tpsDropSetValue('partner', name);
  tps.selection = { city: city, name: partner.name, platform: partner.platform };
  tpsEl('tps-summary').textContent = summary;
  root.classList.toggle('no-connect', !partner.platform);
  // Loading while the selection is saved; nothing is auto-focused when the step switches
  triggers.forEach(function (el) { el.disabled = true; });
  root.classList.add('is-loading');
  root.setAttribute('aria-busy', 'true');
  tpsEl('tps-status').textContent = TPS_TEXT.loading;
  saveTradePartner(tps.selection).then(function () {
    root.classList.remove('is-loading');
    root.setAttribute('aria-busy', 'false');
    tpsEl('tps-status').textContent = TPS_TEXT.selected.replace('{v}', summary);
    triggers.forEach(function (el) { el.disabled = false; });
    tpsSetState('selected');
  });
}
function tpsChange() {
  tps.selection = null;
  tpsEl('tps-status').textContent = '';
  tpsSelectCity('');
  tpsSetState('select');
}
function tpsSetState(state) {
  var root = tpsEl('trade-partner-selector');
  root.dataset.state = state;
  root.querySelectorAll('.ufs-tps__step').forEach(function (step) {
    var show = step.classList.contains('ufs-tps__step--' + state);
    step.hidden = !show;
    step.classList.toggle('is-revealing', show);
  });
}

/* --- Note: Read more / Read less --- */
function tpsToggleNote(expand) {
  var note = tpsEl('tps-note');
  var focusTarget = function () { note.querySelector(expand ? '.ufs-tps__less' : '.ufs-tps__more').focus(); };
  if (note.classList.contains('is-animating')) return;
  if (window.matchMedia('(prefers-reduced-motion: reduce)').matches) { note.classList.toggle('is-expanded', expand); focusTarget(); return; }
  // Measure both end heights, then animate between them. Collapsing keeps the expanded
  // layout until the transition ends, so the text doesn't snap to one line mid-way.
  var from = note.offsetHeight;
  note.classList.toggle('is-expanded', expand);
  var to = note.offsetHeight;
  note.classList.add('is-expanded', expand ? 'is-opening' : 'is-collapsing');
  note.style.height = from + 'px';
  note.offsetHeight;                                   // commit the start height
  note.classList.add('is-animating');
  note.style.height = to + 'px';
  if (expand) { requestAnimationFrame(function () { note.classList.remove('is-opening'); }); focusTarget(); }
  var done = function (e) {
    if (e && (e.target !== note || e.propertyName !== 'height')) return;
    note.removeEventListener('transitionend', done);
    clearTimeout(fallback);
    note.classList.remove('is-animating', 'is-collapsing', 'is-opening');
    note.classList.toggle('is-expanded', expand);
    note.style.height = '';
    if (!expand) focusTarget();
  };
  var fallback = setTimeout(done, 450);
  note.addEventListener('transitionend', done);
}

/* --- Connect drawer --- */
function openTradePartnerPanel(trigger) {
  if (!tps.selection || !tps.selection.platform) return;
  tps.trigger = trigger || document.activeElement;
  document.querySelectorAll('[data-tps-partner]').forEach(function (el) { el.textContent = tps.selection.name; });
  document.querySelectorAll('[data-tps-platform]').forEach(function (el) { el.textContent = tps.selection.platform; });
  var panel = tpsEl('trade-partner-panel');
  panel.inert = false; panel.removeAttribute('aria-hidden');
  tpsEl('trade-partner-backdrop').classList.add('is-open');
  panel.classList.add('is-open');
  document.body.style.overflow = 'hidden';
  tpsEl('trade-partner-panel-body').scrollTop = 0;
  setTimeout(function () { tpsEl('tps-panel-title').focus({ preventScroll: true }); }, 50);
}
function closeTradePartnerPanel() {
  var panel = tpsEl('trade-partner-panel');
  if (!panel.classList.contains('is-open')) return;
  tpsEl('trade-partner-backdrop').classList.remove('is-open');
  panel.classList.remove('is-open');
  panel.inert = true; panel.setAttribute('aria-hidden', 'true');
  document.body.style.overflow = '';
  if (tps.trigger && tps.trigger.offsetParent) tps.trigger.focus({ preventScroll: true });
}
// Connect now: hands over to the partner's platform (SSO / account link), then closes
function connectTradePartner() {
  connectPartnerPlatform(tps.selection);
  closeTradePartnerPanel();
}
// Keep Tab inside an open modal drawer: wrap from the last tabbable element to the
// first and back. Radio groups count once; the focused title (tabindex -1) counts as
// "before the first" so Shift+Tab from it goes to the last.
function trapTab(e, panel) {
  var f = [].filter.call(panel.querySelectorAll('button, input, select, textarea, a[href], [tabindex="0"]'), function (el) {
    if (el.disabled || !el.offsetParent || el.closest('[hidden], [inert]')) return false;
    if (el.type === 'radio' && el.name) {
      var group = panel.querySelectorAll('input[type="radio"][name="' + el.name + '"]');
      var checked = [].filter.call(group, function (r) { return r.checked; })[0];
      return el === (checked || group[0]);
    }
    return true;
  });
  if (!f.length) { e.preventDefault(); return; }
  var first = f[0], last = f[f.length - 1], a = document.activeElement, inList = f.indexOf(a) > -1;
  if (e.shiftKey && (a === first || !inList)) { e.preventDefault(); last.focus(); }
  else if (!e.shiftKey && (a === last || !panel.contains(a))) { e.preventDefault(); first.focus(); }
}
document.addEventListener('keydown', function (e) {
  var panel = tpsEl('trade-partner-panel');
  if (!panel.classList.contains('is-open')) return;
  if (e.key === 'Escape' && !tps.open) closeTradePartnerPanel();
  else if (e.key === 'Tab') trapTab(e, panel);
});
