Overview

Full Component Preview

Recipe ratings: an inline card on the Recipe Detail page that summarises how chefs rated the recipe (average, star display, count and a 5-bar distribution), and a side drawer to add your own. A recipe with no ratings shows a "be the first" prompt instead. Submitting a rating needs a star value and the consent checkbox. The drawer then thanks the user and closes itself after 2 seconds.

    Behavior

    Submitting a Rating

    The rating counts immediately: the inline card behind the drawer updates as soon as the rating is submitted.

    MomentWhat happens
    Submit rating (either state)The drawer opens on a clean form (no star, consent unchecked). Focus moves to the first star.
    Pick starsClick, or arrow keys (→ ↑ up, ← ↓ down; mirrored in RTL). Stars up to the pick light up. Roving tabindex: one Tab stop for the group.
    ConsentRequired: Submit enables only with a star AND the box checked
    Submit RatingSaved; the card re-renders (none → summary on the first rating, bars animate to their new widths); the success step shows "Your rating" with the stars. Focus moves to the success title.
    Auto-closeAfter 2s, with a sweep across the Close button showing the countdown. Pressing Close, the X, the backdrop or Esc closes at once and cancels the timer.
    Close (any way)Focus returns to the Submit rating button that opened it (or the visible one, if the card changed state)
    KeyboardTab / Shift+Tab stay inside the open drawer (wrap from last to first and back).
    JS
    JS
    /* ── Ratings ──
   counts[i] = number of (i + 1)-star ratings for this recipe, from the ratings API.
   The average is shown to one decimal; stars show it exactly (fractional fill). */
var ratingCounts = [0, 0, 0, 0, 0];
var rating = { value: 0, closeTimer: null, trigger: null };
var RATING_TEXT = {
  outOf: '{n} out of 5 stars', barLabel: '{s} star: {c} ratings', count: '({n})'
};
function rt(key, vars) { return RATING_TEXT[key].replace(/\{(\w+)\}/g, function (_, k) { return vars[k]; }); }

function starsMarkup(score) {
  var html = '';
  for (var i = 0; i < 5; i++) html += '<span class="ufs-star" style="--fill:' + Math.max(0, Math.min(1, score - i)) * 100 + '%"></span>';
  return html;
}

function renderRatings() {
  var total = ratingCounts.reduce(function (a, b) { return a + b; }, 0);
  var sum = ratingCounts.reduce(function (a, c, i) { return a + c * (i + 1); }, 0);
  var avg = total ? sum / total : 0;
  document.getElementById('ufs-ratings').dataset.state = total ? 'summary' : 'none';
  document.getElementById('ratings-avg').textContent = avg.toFixed(1);
  var stars = document.getElementById('ratings-avg-stars');
  stars.innerHTML = starsMarkup(avg);
  stars.setAttribute('aria-label', rt('outOf', { n: avg.toFixed(1) }));
  document.getElementById('ratings-count').textContent = rt('count', { n: total });
  var rows = '';
  for (var s = 5; s >= 1; s--) {
    var c = ratingCounts[s - 1];
    rows += '<li class="ufs-ratings__bar-row" aria-label="' + rt('barLabel', { s: s, c: c }) + '">' +
      '<span class="ufs-ratings__bar-label" aria-hidden="true">' + s + '<span class="ufs-star"></span></span>' +
      '<span class="ufs-ratings__bar" aria-hidden="true"><span class="ufs-ratings__bar-fill" style="width:' + (total ? c / total * 100 : 0) + '%"></span></span>' +
      '<span class="ufs-ratings__bar-count" aria-hidden="true">' + c + '</span></li>';
  }
  document.getElementById('ratings-bars').innerHTML = rows;
}

/* Star input: radio group with a roving tabindex. Click / Enter / Space picks;
   arrow keys move and pick (mirrored in RTL); stars up to the pick light up */
function setRatingValue(v, focus) {
  rating.value = v;
  document.querySelectorAll('.ufs-rating-input__star').forEach(function (btn) {
    var n = +btn.dataset.value;
    btn.classList.toggle('is-on', n <= v);
    btn.setAttribute('aria-checked', n === v ? 'true' : 'false');
    btn.tabIndex = n === (v || 1) ? 0 : -1;
    if (focus && n === v) btn.focus();
  });
  updateRatingSubmit();
}
document.querySelectorAll('.ufs-rating-input__star').forEach(function (btn) {
  btn.addEventListener('click', function () { setRatingValue(+btn.dataset.value); });
  btn.addEventListener('keydown', function (e) {
    var rtl = getComputedStyle(btn).direction === 'rtl';
    var d = { ArrowRight: rtl ? -1 : 1, ArrowUp: 1, ArrowLeft: rtl ? 1 : -1, ArrowDown: -1 }[e.key];
    if (!d) return;
    e.preventDefault();
    setRatingValue(Math.min(5, Math.max(1, (rating.value || 0) + d)), true);   // from no rating, → picks 1 star
  });
});

// Submit needs a rating AND the consent box
function updateRatingSubmit() {
  document.getElementById('rating-submit').disabled = !(rating.value > 0 && document.getElementById('rating-consent').checked);
}

function openRatingPanel(trigger) {
  rating.trigger = trigger || document.activeElement;
  setRatingValue(0);
  document.getElementById('rating-consent').checked = false;
  updateRatingSubmit();
  document.getElementById('rating-step-form').hidden = false;
  document.getElementById('rating-step-success').hidden = true;
  document.getElementById('rating-close-btn').classList.remove('is-counting');
  var panel = document.getElementById('rating-panel');
  panel.inert = false; panel.removeAttribute('aria-hidden');

  panel.classList.add('is-open');

function closeRatingPanel() {
  clearTimeout(rating.closeTimer);
  var panel = document.getElementById('rating-panel');
  if (!panel.classList.contains('is-open')) return;
  document.getElementById('rating-backdrop').classList.remove('is-open');
  panel.classList.remove('is-open');
  panel.inert = true; panel.setAttribute('aria-hidden', 'true');
  document.body.style.overflow = '';
  if (rating.trigger && document.contains(rating.trigger) && rating.trigger.offsetParent) rating.trigger.focus({ preventScroll: true });
  else [].filter.call(document.querySelectorAll('#ufs-ratings .btn'), function (b) { return b.offsetParent; })[0].focus({ preventScroll: true });   // the card switched state: its visible button
}
// Keep Tab inside an open modal drawer: wrap from the last tabbable element to the
// first and back. Radio groups count once; the focused title (tabindex -1) counts as
// "before the first" so Shift+Tab from it goes to the last.
function trapTab(e, panel) {
  var f = [].filter.call(panel.querySelectorAll('button, input, select, textarea, a[href], [tabindex="0"]'), function (el) {
    if (el.disabled || !el.offsetParent || el.closest('[hidden], [inert]')) return false;
    if (el.type === 'radio' && el.name) {
      var group = panel.querySelectorAll('input[type="radio"][name="' + el.name + '"]');
      var checked = [].filter.call(group, function (r) { return r.checked; })[0];
      return el === (checked || group[0]);
    }
    return true;
  });
  if (!f.length) { e.preventDefault(); return; }
  var first = f[0], last = f[f.length - 1], a = document.activeElement, inList = f.indexOf(a) > -1;
  if (e.shiftKey && (a === first || !inList)) { e.preventDefault(); last.focus(); }
  else if (!e.shiftKey && (a === last || !panel.contains(a))) { e.preventDefault(); first.focus(); }
}
document.addEventListener('keydown', function (e) {
  var panel = document.getElementById('rating-panel');
  if (!panel.classList.contains('is-open')) return;
  if (e.key === 'Escape') closeRatingPanel();
  else if (e.key === 'Tab') trapTab(e, panel);
});

function submitRating() {
  if (!rating.value || !document.getElementById('rating-consent').checked) return;
  saveRating(rating.value);                                  // POST to the ratings API
  ratingCounts[rating.value - 1]++;
  renderRatings();                                           // the inline card updates behind the drawer
  var result = document.getElementById('rating-result-stars');
  result.innerHTML = starsMarkup(rating.value);
  result.setAttribute('aria-label', rt('outOf', { n: rating.value }));
  document.getElementById('rating-step-form').hidden = true;
  document.getElementById('rating-step-success').hidden = false;
  document.getElementById('rating-panel-body').scrollTop = 0;
  document.getElementById('rating-success-title').focus({ preventScroll: true });
  // Auto-close after 2s; the Close button's sweep shows the countdown
  var btn = document.getElementById('rating-close-btn');
  btn.classList.remove('is-counting'); void btn.offsetWidth; btn.classList.add('is-counting');
  clearTimeout(rating.closeTimer);
  rating.closeTimer = setTimeout(closeRatingPanel, 2000);
}
