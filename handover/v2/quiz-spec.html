Quiz

Multiple-choice quiz at the end of a training, in the v1 side drawer: check each answer, see right and wrong straight away, pass at 80% to download a certificate.

Overview

Full Component Preview

A short multiple-choice quiz at the end of a training. Passing it (80%) earns a certificate. It opens from a Promo Banner on the Training Detail page, in the v1 side drawer. Each question is answered and checked before moving on: the buyer sees straight away whether they were right, and which answer was. The drawer then ends on a pass or fail screen. Taking the quiz needs sign-in: signed-out visitors get the v1 sign-in drawer first, with quiz copy, and the quiz opens as soon as they are signed in.

💡
Live: the banner's "Take the quiz" runs the real 5-question quiz (correct answers: 3, 3, 2, 1, 2). Get two wrong to see the fail screen.

Behavior

Flow & Scoring

One button drives the whole run. The score only counts checked first picks: an option can't be changed after "Check the answer".

MomentWhat happens
OpenSigned in: drawer opens, attempt starts at question 1, focus on the question. Signed out: v1 sign-in drawer with "Sign up or Log in to take the quiz" / "Pass it to earn a free certificate.", then the quiz opens by itself after sign-in.
Pick an optionEnables "Check the answer"
Check the answerLocks the options, marks the right answer (and the wrong pick), announces the result, button becomes "Next question" / "See results"
Next questionProgress fills one more step, the next question renders, focus moves to it, the panel scrolls to the top
See resultsscore ÷ questions ≥ 0.8 → pass view, otherwise fail. Focus moves to the result title.
Fail"Re-watch videos" closes the drawer and scrolls to the training's Course Content. "Re-take quiz" restarts at question 1.
Pass"Download Certificate" spins (v1 .btn-loading) while the PDF is generated, then reads "Certificate downloaded" and disables. "Continue Learning" closes.
Close / backdrop / EscCloses and throws the attempt away. Reopening always starts again. Focus returns to the trigger.
KeyboardTab / Shift+Tab stay inside the open drawer (wrap from last to first and back).

var quiz = { step: 0, score: 0, checked: false, trigger: null };

function quizEsc(s) { var d = document.createElement('div'); d.textContent = s; return d.innerHTML; }
function quizShow(view) {
  document.getElementById('quiz-run').hidden = view !== 'run';
  document.getElementById('quiz-result-fail').hidden = view !== 'fail';
  document.getElementById('quiz-result-pass').hidden = view !== 'pass';
  document.getElementById('quiz-panel-body').scrollTop = 0;
}
function renderQuizQuestion() {
  var item = QUIZ[quiz.step], total = QUIZ.length;
  quiz.checked = false;
  document.getElementById('quiz-fill').style.width = ((quiz.step + 1) / total * 100) + '%';
  var track = document.getElementById('quiz-track');
  track.setAttribute('aria-valuemax', total);
  track.setAttribute('aria-valuenow', quiz.step + 1);
  document.getElementById('quiz-count').textContent = qt('count', { n: quiz.step + 1, total: total });
  document.getElementById('quiz-question').textContent = item.q;
  var opts = document.getElementById('quiz-options');
  opts.classList.remove('is-checked');
  opts.innerHTML = item.options.map(function (o, i) {
    var media = item.type === 'text' ? '<span>' + quizEsc(o.text) + '</span>'
      : '<span class="ufs-quiz__media' + (item.type === 'image' ? ' ufs-quiz__media--image-only' : '') + '">' +
          '<img class="ufs-quiz__img" src="' + QUIZ_IMG + o.img + '" alt="' + (item.type === 'image' ? quizEsc(o.text) : '') + '"/>' +
          (item.type === 'image' ? '' : '<span>' + quizEsc(o.text) + '</span>') + '</span>';
    return '<label class="ufs-quiz__option"><input type="radio" name="quiz-option" value="' + i + '" onchange="quizSelect()"/>' +
      '<span class="ufs-quiz__radio" aria-hidden="true"></span>' + MARK_WRONG + MARK_RIGHT + media + '</label>';
  }).join('');
  var btn = document.getElementById('quiz-action');
  btn.textContent = qt('check', {});
  btn.disabled = true;                              // until an option is picked
}
function quizSelect() { if (!quiz.checked) document.getElementById('quiz-action').disabled = false; }

// One button, two jobs: check the picked answer, then move on
function quizAction() {
  var item = QUIZ[quiz.step], picked = document.querySelector('#quiz-options input:checked');
  if (!quiz.checked) {
    if (!picked) return;
    var pick = +picked.value, labels = document.querySelectorAll('#quiz-options .ufs-quiz__option');
    quiz.checked = true;
    if (pick === item.answer) quiz.score++;
    labels[item.answer].classList.add('is-right');
    if (pick !== item.answer) labels[pick].classList.add('is-wrong');
    document.getElementById('quiz-options').classList.add('is-checked');
    document.querySelectorAll('#quiz-options input').forEach(function (i) { i.disabled = true; });
    document.getElementById('quiz-status').textContent = pick === item.answer
      ? qt('correct', {}) : qt('incorrect', { answer: item.options[item.answer].text });
    document.getElementById('quiz-action').textContent = qt(quiz.step === QUIZ.length - 1 ? 'results' : 'next', {});
    return;
  }
  if (quiz.step < QUIZ.length - 1) {
    quiz.step++;
    renderQuizQuestion();
    document.getElementById('quiz-question').focus({ preventScroll: true });
    return;
  }
  var passed = quiz.score / QUIZ.length >= QUIZ_PASS;
  quizShow(passed ? 'pass' : 'fail');
  document.querySelector('#quiz-result-' + (passed ? 'pass' : 'fail') + ' .ufs-quiz__result-title').focus({ preventScroll: true });
}

function startQuiz() {
  quiz.step = 0; quiz.score = 0;
  var dl = document.getElementById('quiz-download');
  dl.classList.remove('btn-loading'); dl.disabled = false; dl.textContent = qt('download', {});
  document.getElementById('quiz-status').textContent = '';
  renderQuizQuestion();
  quizShow('run');
  document.getElementById('quiz-question').focus({ preventScroll: true });
}
// Signed out: the v1 sign-in drawer opens first with the 'quiz' copy, and the quiz opens straight after sign-in
function openQuiz(trigger) {
  if (!isSignedIn()) { openSignIn('quiz', function () { openQuiz(trigger); }); return; }
  quiz.trigger = trigger || document.activeElement;
  var panel = document.getElementById('quiz-panel');
  panel.inert = false; panel.removeAttribute('aria-hidden');
  document.getElementById('quiz-backdrop').classList.add('is-open');
  panel.classList.add('is-open');
  document.body.style.overflow = 'hidden';
  setTimeout(startQuiz, 50);
}
function closeQuizPanel() {
  var panel = document.getElementById('quiz-panel');
  if (!panel.classList.contains('is-open')) return;
  document.getElementById('quiz-backdrop').classList.remove('is-open');
  panel.classList.remove('is-open');
  panel.inert = true; panel.setAttribute('aria-hidden', 'true');
  document.body.style.overflow = '';
  if (quiz.trigger && document.contains(quiz.trigger)) quiz.trigger.focus({ preventScroll: true });
}
// Keep Tab inside an open modal drawer: wrap from the last tabbable element to the
// first and back. Radio groups count once; the focused title (tabindex -1) counts as
// "before the first" so Shift+Tab from it goes to the last.
function trapTab(e, panel) {
  var f = [].filter.call(panel.querySelectorAll('button, input, select, textarea, a[href], [tabindex="0"]'), function (el) {
    if (el.disabled || !el.offsetParent || el.closest('[hidden], [inert]')) return false;
    if (el.type === 'radio' && el.name) {
      var group = panel.querySelectorAll('input[type="radio"][name="' + el.name + '"]');
      var checked = [].filter.call(group, function (r) { return r.checked; })[0];
      return el === (checked || group[0]);
    }
    return true;
  });
  if (!f.length) { e.preventDefault(); return; }
  var first = f[0], last = f[f.length - 1], a = document.activeElement, inList = f.indexOf(a) > -1;
  if (e.shiftKey && (a === first || !inList)) { e.preventDefault(); last.focus(); }
  else if (!e.shiftKey && (a === last || !panel.contains(a))) { e.preventDefault(); first.focus(); }
}
document.addEventListener('keydown', function (e) {
  var panel = document.getElementById('quiz-panel');
  if (!panel.classList.contains('is-open')) return;
  if (e.key === 'Escape') closeQuizPanel();
  else if (e.key === 'Tab') trapTab(e, panel);
});

// Fail: back to the training's videos (Course Content section)
function quizRewatch() {
  closeQuizPanel();
  document.getElementById('training-course-content').scrollIntoView({ behavior: 'smooth', block: 'start' });
}
// Pass: the button spins while the certificate PDF is generated, then confirms
function quizDownload() {
  var btn = document.getElementById('quiz-download');
  if (btn.disabled) return;
  btn.classList.add('btn-loading');
  requestCertificate().then(function () {
    btn.classList.remove('btn-loading');
    btn.textContent = qt('downloaded', {});
    btn.disabled = true;
  });
}
