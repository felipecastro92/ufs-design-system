Wholesaler Picker

Side drawer opened by "Buy it now" on a product tile (non-webshop markets): searchable list of wholesalers, each linking to its site, with expandable multi-option wholesalers.

Source
handover/
template-products-catalog.html
Format·HTML + CSS + JS
Overview

Full Component Preview

Outside the webshop, "Buy it now" on a product tile opens the Wholesaler Picker: a side drawer listing the wholesalers that stock the product in this market. Each row sends the chef to that wholesaler's site. A wholesaler with several ordering options (Bidfood in the example) expands to show them. A search field filters the list. In webshop markets this flow is replaced by the Trade Partner Selector and the cart.

💡
Live example in context: v1 Products Catalog template → "Buy it now" on any tile.
Live

Doc stand-in for the v1 Product Tile CTA (.ufs-tile__cta). Try searching "direct" or "xyz".

Structure

Anatomy

Inside the v1 drawer shell: title, description, search, then .ufs-wholesaler-list. Every wholesaler is its own bordered card. A single-option wholesaler is one link (a.ufs-wholesaler-item.ufs-wholesaler-row); a multi-option one is a v1 accordion item whose content holds the sub-option links.

Choosing a Wholesaler

The list, logos, prices and options come from the CMS per market and per product. There is no fixed count: the example has five.

/* ── Wholesaler Picker ── */
var wholesaler = { trigger: null };
function openWholesalerPanel(trigger) {
  wholesaler.trigger = trigger || document.activeElement;
  var panel = document.getElementById('wholesaler-panel');
  document.getElementById('wholesaler-search').value = '';
  filterWholesalers('');
  panel.inert = false; panel.removeAttribute('aria-hidden');
  document.getElementById('wholesaler-backdrop').classList.add('is-open');
  panel.classList.add('is-open');
  document.body.style.overflow = 'hidden';
  document.getElementById('wholesaler-panel-body').scrollTop = 0;
  // The title, not the search: focusing the search would pop the keyboard on mobile
  setTimeout(function () { document.getElementById('wholesaler-panel-title').focus({ preventScroll: true }); }, 50);
}
function closeWholesalerPanel() {
  var panel = document.getElementById('wholesaler-panel');
  if (!panel.classList.contains('is-open')) return;
  document.getElementById('wholesaler-backdrop').classList.remove('is-open');
  panel.classList.remove('is-open');
  panel.inert = true; panel.setAttribute('aria-hidden', 'true');
  document.body.style.overflow = '';
  if (wholesaler.trigger && wholesaler.trigger.offsetParent) wholesaler.trigger.focus({ preventScroll: true });
}
// Keep Tab inside an open modal drawer: wrap from the last tabbable element to the
// first and back. Radio groups count once; the focused title (tabindex -1) counts as
// "before the first" so Shift+Tab from it goes to the last.
function trapTab(e, panel) {
  var f = [].filter.call(panel.querySelectorAll('button, input, select, textarea, a[href], [tabindex="0"]'), function (el) {
    if (el.disabled || !el.offsetParent || el.closest('[hidden], [inert]')) return false;
    if (el.type === 'radio' && el.name) {
      var group = panel.querySelectorAll('input[type="radio"][name="' + el.name + '"]');
      var checked = [].filter.call(group, function (r) { return r.checked; })[0];
      return el === (checked || group[0]);
    }
    return true;
  });
  if (!f.length) { e.preventDefault(); return; }
  var first = f[0], last = f[f.length - 1], a = document.activeElement, inList = f.indexOf(a) > -1;
  if (e.shiftKey && (a === first || !inList)) { e.preventDefault(); last.focus(); }
  else if (!e.shiftKey && (a === last || !panel.contains(a))) { e.preventDefault(); first.focus(); }
}
document.addEventListener('keydown', function (e) {
  var panel = document.getElementById('wholesaler-panel');
  if (!panel.classList.contains('is-open')) return;
  if (e.key === 'Escape') closeWholesalerPanel();
  else if (e.key === 'Tab') trapTab(e, panel);
});

/* Matches each card's data-wholesaler-name (its own name plus any sub-option names,
   e.g. "Bidfood Bidfood Direct Bidfood Wholesaler") against the query. A sub-option
   match keeps its parent visible and expands it, so "direct" surfaces Bidfood. */
function filterWholesalers(query) {
  var q = query.trim().toLowerCase();
  var list = document.getElementById('wholesaler-list');
  var anyVisible = false;
  list.querySelectorAll('.ufs-wholesaler-item').forEach(function (item) {
    var matches = !q || (item.dataset.wholesalerName || '').toLowerCase().indexOf(q) !== -1;
    item.style.display = matches ? '' : 'none';
    if (matches) anyVisible = true;
    var trigger = item.querySelector('.ufs-accordion__trigger');
    if (matches && trigger && q) {
      var nameOnly = item.querySelector('.ufs-wholesaler-name').textContent.toLowerCase();
      if (nameOnly.indexOf(q) === -1 && !item.classList.contains('is-open')) toggleAccordionItem(trigger);
    }
  });
  list.classList.toggle('is-empty', !anyVisible);
}

/* v1 accordion toggle (shared with the FAQ): one item open per group */
function toggleAccordionItem(trigger) {
  var item = trigger.closest('.ufs-accordion__item'), panel = trigger.closest('.ufs-accordion__panel');
  var isOpen = item.classList.contains('is-open');
  panel.querySelectorAll('.ufs-accordion__item.is-open').forEach(function (i) {
    i.classList.remove('is-open');
    i.querySelector('.ufs-accordion__trigger').setAttribute('aria-expanded', 'false');
  });
  if (!isOpen) { item.classList.add('is-open'); trigger.setAttribute('aria-expanded', 'true'); }
}
