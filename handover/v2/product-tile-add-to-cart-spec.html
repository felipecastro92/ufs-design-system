Product Tile + Add to Cart

Webshop Product Tile: points, price, pack sizes and an Add to cart that becomes the product's cart state. Multi-pack products open the Pack Size Picker.

Overview

Full Component Preview

The webshop version of the v1 Product Tile (tiles-spec.html). It adds a points pill, a price line, the available pack sizes and an Add to cart button that turns into the line's cart state. Everything else (photo, favourite, eyebrow, title, hover, badge) is the v1 tile, unchanged. Products with a multi-pack open the Pack Size Picker. Single-item-only products get an inline stepper instead.

💡
Live: the first two tiles have a Pack of 6 (Add to cart → picker). The third is sold as single items only (Add to cart → inline stepper). The picker, toasts, fly-to-cart and badge are the real shared components.

Structure

Anatomy

Same anatomy as the v1 Product Tile, plus the modifier .ufs-tile--shop. The rating and the tile-wide CTA link are dropped. The whole tile links to the product page through a stretched title link, and the cart controls sit above it.

Behavior

Add to Cart Flow

Pressing Add to cart always puts 1 single item in the cart immediately, so one click is enough for the common case. What happens next depends on the product.

Flow
ProductOn "Add to cart"Focus
Has a multi-pack1 single added → photo flies to the cart → the picker opens anchored to this tile (trigger = the new Edit link) to adjust both packsPicker dialog; back to Edit on close
Single items only1 single added → photo flies → the CTA becomes the inline stepper+ of the stepper (not the field, so no selection ring appears after a mouse click)
Stepper / Edit laterEvery change is applied immediately and toasted ("Updated: … × 3")Stays on the pressed button. At 0, moves to Add to cart.
JS

JS
/* ── Product Tile + Add to cart ──
   Uses the shared cart engine + Pack Size Picker (pack-size-picker-spec.html). */
TEXT.packInfo = '<b>Single {unit}</b> or <b>Pack of {n}</b>';
TEXT.packInfoSingle = '<b>Single {unit}</b>';
TEXT.inCartParts = 'In cart: {parts}';
TEXT.partSingle = '{q} single';
TEXT.partMulti = '{q} × pack of {n}';

function tileCtaHTML(p) {
  var l = lineOf(p.id);
  // Nothing in the cart: the "Add to cart" button
  if (!lineUnits(p, l)) {
    return '<button type="button" class="btn btn-primary" data-picker-return="' + p.id + '" data-add onclick="tileAdd(\'' + p.id + '\')">' +
      '<img src="../icons/cart.svg" alt=""/>' + t('add') + '</button>';
  }
  // Sold as single items only: an inline stepper, no picker
  if (!p.multi) {
    return '<div class="ufs-tile__qty"><span class="ufs-tile__qty-label">' + t('inCart') + '</span>' +
      stepperHTML({ p: p, pack: 'single', value: l.single, scope: 'tile', args: "'" + p.id + "','single'" }) + '</div>';
  }
  // Has a multi-pack: what's in the cart + Edit (reopens the picker)
  var parts = [];
  if (l.single) parts.push(t('partSingle', { q: l.single }));
  if (l.multi) parts.push(t('partMulti', { q: l.multi, n: p.multi }));
  return '<div class="ufs-tile__incart"><span class="ufs-tile__incart-text">' + ICON.check +
      '<span>' + t('inCartParts', { parts: parts.join(' · ') }) + '</span></span>' +
    '<button type="button" class="ufs-tile__incart-edit" data-picker-return="' + p.id + '"' +
      ' onclick="openPicker(\'' + p.id + '\', this)" aria-label="' + escHTML(t('edit') + ' ' + p.name) + '">' + t('edit') + '</button></div>';
}

function renderTileCta(id) {
  var cta = document.getElementById('cta-' + id);
  if (cta) cta.innerHTML = tileCtaHTML(PRODUCTS[id]);
}
cartViews.push(renderTileCta);          // picker and cart panel edits keep the tile in sync

function tilePhoto(id) { return document.querySelector('#tile-' + id + ' .ufs-tile__image-media'); }

// "Add to cart": 1 single item goes in straight away. Multi-pack products then open
// the picker to adjust; single-only products land on + of the inline stepper.
function tileAdd(id) {
  var p = PRODUCTS[id];
  setLine(id, 'single', 1);
  flyToCart(tilePhoto(id), IMG + p.img);
  if (p.multi) openPicker(id, document.querySelector('#cta-' + id + ' .ufs-tile__incart-edit'));
  else document.querySelector('#cta-' + id + ' [data-role="inc"]').focus({ preventScroll: true });
}

// Inline stepper (single-only products)
function tileStep(id, pack, d) { tileSet(id, pack, (lineOf(id)[pack] || 0) + d, true); }
function tileSet(id, pack, value, fromStep) {
  var before = setLine(id, pack, value), q = lineOf(id)[pack] || 0, cta = document.getElementById('cta-' + id);
  if (q === before) { renderTileCta(id); return; }
  if (!q) { bumpBadge(); cta.querySelector('[data-add]').focus({ preventScroll: true }); return; }   // back to "Add to cart"
  rollQty(cta.querySelector('.fi-qty'), before, q > before);
  if (q > before) flyToCart(tilePhoto(id), IMG + PRODUCTS[id].img); else bumpBadge();
  if (fromStep) cta.querySelector(q < before ? '[data-role="dec"]' : '[data-role="inc"]').focus({ preventScroll: true });
}
ℹ️
The tile itself is a link to the product page. The cart controls and the favourite button sit above the stretched link (z-index: 2). The points pill and badge let clicks through (pointer-events: none), so tapping them opens the product.

Internationalisation

i18n & RTL
