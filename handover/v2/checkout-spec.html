Overview

Flow & Layout

The webshop checkout is one page with five steps, hash-routed so the browser Back button works. It ends in an order request to the buyer's trade partner, not a payment: the partner confirms the final prices, payment and delivery. Steps 2–4 run inside the v1 Checkout Header (secure badge, "Back to cart", named-step progress, mobile numeric bar and collapsible order summary), with 3 named steps: Details · Delivery · Review. Steps 1 and 5 keep the regular site header.

Routing JS
You can't skip ahead: a deep link to a later step lands on the first step that still needs input. Leaving the confirmation resets the flow.
JS
/* ── Routing: one page, five steps, hash-routed so browser Back works ──
   #cart → #details → #delivery → #review → #success.
   Steps 2–4 switch the header to the v1 checkout context (body.ctx-checkout:
   progress bar, "Back to cart", mobile order summary). 1 and 5 keep the site header. */
var co = { started: false, signedIn: false, done: {}, when: 'asap', date: null, order: null };
var STEPS = ['cart', 'details', 'delivery', 'review', 'success'];
var FLOW = [{ id: 'details', label: 'Details' }, { id: 'delivery', label: 'Delivery' }, { id: 'review', label: 'Review' }];

function goTo(step) { if (location.hash !== '#' + step) location.hash = step; else route(); }
function route() {
  var step = location.hash.replace('#', '') || 'cart';
  if (STEPS.indexOf(step) < 0) step = 'cart';
  // No skipping ahead: land on the first step that still needs input
  if (step === 'success' && !co.order) step = 'cart';
  if (step !== 'cart' && step !== 'success' && !cartLines().length) step = 'cart';
  if (step === 'details' && !co.started) step = 'cart';
  if (step === 'delivery' && !co.done.details) step = co.started ? 'details' : 'cart';
  if (step === 'review' && !co.done.delivery) step = co.done.details ? 'delivery' : co.started ? 'details' : 'cart';
  if (co.order && step !== 'success') co.order = null;          // left the confirmation: start over
  if (location.hash && '#' + step !== location.hash) history.replaceState(null, '', '#' + step);

  var flow = step === 'details' || step === 'delivery' || step === 'review';
  document.body.classList.toggle('ctx-checkout', flow);
  if (flow) renderProgress(step);
  document.querySelectorAll('.ufs-checkout-step').forEach(function (s) { s.hidden = s.getAttribute('data-step') !== step; });
  renderSummaries(false);
  ({ details: enterDetails, delivery: enterDelivery, review: renderReview, success: renderSuccess }[step] || function () {})();
  window.scrollTo(0, 0);
  // Move focus to the step's heading (not on first load: that would steal focus from the skip link)
  var h = document.querySelector('#step-' + step + ' .ufs-checkout__title');
  if (h && route.ran) h.focus({ preventScroll: true });
  route.ran = true;
}
window.addEventListener('hashchange', route);

// v1 checkout header: named steps (desktop) + numeric bar (mobile)
function renderProgress(step) {
  var cur = FLOW.map(function (f) { return f.id; }).indexOf(step);
  document.getElementById('checkout-progress-desktop').innerHTML = FLOW.map(function (f, i) {
    var cls = i < cur ? ' is-done' : i === cur ? ' is-current' : '';
    return (i ? '<div class="ufs-checkout-progress__connector' + (i <= cur ? ' is-done' : '') + '"></div>' : '') +
      '<div class="ufs-checkout-progress__step' + cls + '"' + (i === cur ? ' aria-current="step"' : '') + '>' +
        '<span class="ufs-checkout-progress__dot">' + (i < cur ? ICON.check : i + 1) + '</span>' +
        '<span class="ufs-checkout-progress__label">' + f.label + '</span></div>';
  }).join('');
  document.getElementById('checkout-progress-mobile').innerHTML =
    '<span class="ufs-form-progress__label">Step ' + (cur + 1) + ' of 3 · ' + FLOW[cur].label + '</span><div class="ufs-form-progress__track">' +
    FLOW.map(function (f, i) { return '<div class="ufs-form-progress__step' + (i < cur ? ' is-done' : i === cur ? ' is-current' : '') + '"></div>'; }).join('') + '</div>';
}

// Step 1 → 2. Signing in pre-fills the details; guests start empty.
function startCheckout(signedIn) {
  if (!co.started || signedIn) prefill(signedIn);
  co.started = true;
  goTo('details');
}

    /* ── Totals + discounts (cart lines come from the Cart Side Panel's cartLines()) ── */
function totals() {
  var lines = cartLines(), sub = lines.reduce(function (a, r) { return a + r.price; }, 0), disc = discounts(sub);
  return {
    lines: lines, sub: sub, disc: disc, total: Math.max(0, sub - disc.total),
    count: lines.reduce(function (a, r) { return a + r.q; }, 0),
    pts: lines.reduce(function (a, r) { return a + r.q * (r.pack === 'multi' ? r.p.multi : 1) * r.p.pts; }, 0)
  };
}

function discounts(sub) {
  var left = sub, lines = [];
  var ordered = appliedCodes.filter(function (c) { return CODES[c].kind === 'voucher'; })
    .concat(appliedCodes.filter(function (c) { return CODES[c].kind === 'gift'; }));
  ordered.forEach(function (c) {
    var d = CODES[c], amt = 0, note = d.label;
    if (d.kind === 'voucher' && d.min && sub < d.min) note = 'Spend ' + money(d.min - sub) + ' more to use this code';   // kept, but worth €0 for now
    else amt = Math.min(left, d.pct ? round2(sub * d.pct / 100) : d.off);
    left -= amt;
    lines.push({ code: c, kind: d.kind, note: note, amount: amt });
  });
  return { lines: lines, total: sub - left };
}

function applyCode(e) {
  if (e) e.preventDefault();
  var input = $('promo-code'), code = input.value.trim().toUpperCase().replace(/\s+/g, ''), d = CODES[code], t = totals();
  if (!code) return false;
  var voucher = appliedCodes.filter(function (c) { return CODES[c].kind === 'voucher'; })[0];
  if (!d) codeMsg("This code isn't valid. Check it and try again.");
  else if (appliedCodes.indexOf(code) !== -1) codeMsg('This code is already applied.');
  else if (d.expired) codeMsg('This code expired on ' + d.expired + '.');
  else if (d.kind === 'voucher' && voucher) codeMsg('Only one voucher per order. Remove ' + voucher + ' to use this one.');
  else if (d.min && t.sub < d.min) codeMsg('Spend ' + money(d.min) + ' or more to use this code (' + money(d.min - t.sub) + ' to go).');
  else if (d.kind === 'gift' && t.total <= 0) codeMsg('Your order total is already ' + money(0) + '.');
  else {
    appliedCodes.push(code);
    input.value = ''; codeInput();
    renderSummaries(false);
    codeMsg((d.kind === 'gift' ? 'Gift card ' : 'Voucher ') + code + ' applied: ' + d.label + '.', true);
  }
  input.focus({ preventScroll: true });
  return false;
}
function removeCode(code) {
  appliedCodes = appliedCodes.filter(function (c) { return c !== code; });
  renderSummaries(false);
  codeMsg(code + ' removed.', true);
  toggleCodes(true);
}

function ymd(d) { return d.getFullYear() + '-' + ('0' + (d.getMonth() + 1)).slice(-2) + '-' + ('0' + d.getDate()).slice(-2); }
function fromKey(k) { var x = k.split('-'); return new Date(+x[0], x[1] - 1, +x[2], 12); }
function addDays(d, n) { var x = new Date(d); x.setDate(x.getDate() + n); return x; }
function today() { var d = new Date(); d.setHours(12, 0, 0, 0); return d; }
function calMin() { return addDays(today(), 2); }

function fullKeys() {        // demo: the 3rd and 8th available weekdays are fully booked
  var out = [], d = calMin(), n = 0;
  while (out.length < 2) { if (d.getDay() % 6) { n++; if (n === 3 || n === 8) out.push(ymd(d)); } d = addDays(d, 1); }
  return out;
}
function dayState(d) {
  if (d < calMin() || d > calMax()) return 'out';
  if (!(d.getDay() % 6)) return 'weekend';
  return fullKeys().indexOf(ymd(d)) !== -1 ? 'full' : 'ok';
}

function setError(wrap, msg) {
  wrap.classList.toggle('is-error', !!msg);
  var m = wrap.querySelector('.fi-error-msg'), f = wrap.querySelector('.fi-field');
  if (!m && msg) { m = document.createElement('p'); m.className = 'fi-error-msg'; m.id = 'err-' + f.id; wrap.appendChild(m); }
  if (!m) return;
  if (msg) m.textContent = msg;
  m.hidden = !msg;
  if (msg) { f.setAttribute('aria-invalid', 'true'); f.setAttribute('aria-describedby', m.id); } else f.removeAttribute('aria-invalid');
}
function checkField(wrap) {
  var f = wrap.querySelector('.fi-field'), v = f.value.trim(), type = wrap.getAttribute('data-type');
  var label = wrap.querySelector('.fi-label').textContent.replace(/\s*\(optional\)/, '').toLowerCase();
  if (!v) return f.tagName === 'SELECT' ? 'Select a ' + label : 'Enter your ' + label;
  if (type === 'email' && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(v)) return 'Enter a valid email, e.g. name@business.co.za';
  if (type === 'phone' && v.replace(/\D/g, '').length < 9) return 'Enter a valid phone number';
  if (type === 'postal' && !/^\d{4}$/.test(v)) return 'Enter a 4-digit postal code';
  return '';
}
function validate(form) {        // returns the first invalid field (or null)
  var first = null;
  form.querySelectorAll('.fi-wrap[data-req]').forEach(function (w) {

    setError(w, msg);
    if (msg && !first) first = w.querySelector('.fi-field');
  });
  return first;
}
['focusout', 'change'].forEach(function (type) {
