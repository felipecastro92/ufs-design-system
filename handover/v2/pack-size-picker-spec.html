Pack Size Picker

Popover (desktop) / bottom sheet (mobile) for choosing how many single items and multi-packs of one product to put in the cart. Edits the cart live, with points and a running total.

Overview

Full Component Preview

Lets a buyer set quantities for every pack type of one product in one place: Single item and Pack of N. It opens from the Product Tile's Add to cart (and its Edit link) as a popover on desktop and a bottom sheet on mobile. Changes go straight into the cart, so there is no confirm button. The Product Hero uses the same pack rows inline instead of in a popover. See product-hero-add-to-cart-spec.html.

Structure

Anatomy

One dialog per page (#pack-picker), refilled for whichever product opened it. It contains a header, two pack rows and a summary. The arrow only exists on desktop. On mobile the sheet gets a grab handle instead (a ::before, purely visual).

Parts
  1. 1 · Arrow: __arrow, points at the trigger (desktop only)
  2. 2 · Title: __title, the product name (Body, 600)
  3. 3 · Subtitle: __sub, "Choose pack size and quantity"
  4. 4 · Close: __close, 36px round button
  5. 5 · Pack rows: .ufs-pack-row ×2, Single item then Pack of N
  6. 6 · Summary: .ufs-pack-summary, items in cart + total price
  7. 7 · Points: .ufs-points, the live points total for this product
Reference markup
The rows and summary values are rendered by renderPicker() (see Behavior). The shell is static.

Structure

Pack Row & Summary

A pack row is the shared building block of the picker and the Product Hero. It holds a thumbnail (with a ×N count on the multi-pack), the pack name, a price line and a control. Each product has exactly two rows. If a product is sold as single items only, it never opens the picker: its tile adds directly (see the tile spec).

Values
All amounts are derived from three product fields: price (one single item), multi (items per pack, or null) and pts (points per single item).
ValueFormulaExample (Tomato Pronto: €18.90, multi 6, 350 pts)
Single item priceprice€18.90
Pack priceprice × multi€113.40
Items in cartsingle + multi × N1 single + 1 pack → 7 items
Totalsingle × price + multi × price × N€132.30
Pointsitems × pts7 × 350 = 2.450

Behavior

Opening & Placement

The picker opens in two situations. (1) A tile's Add to cart is pressed on a product that has a multi-pack: 1 single item goes straight into the cart, then the picker opens so the buyer can adjust. (2) The tile's Edit link is pressed. Only one picker is open at a time, and opening the cart panel closes it.

Placement
ViewportBehaviour
≥769pxPopover, always above the trigger (12px gap), with the arrow on its bottom edge centred on the trigger. It never opens below. If there isn't room above, the page first smooth-scrolls by just enough for the picker to sit 16px below the sticky site header, with the picker moving along with the tile (instant with reduced motion) (mark it with data-sticky-header; falls back to the viewport top). Width and left edge match the anchoring element ([data-picker-anchor], the tile). It repositions on resize and closes when the user scrolls (its own scroll-to-fit doesn't count).
≤768pxBottom sheet, full width, slides up over a 50% black backdrop. The width and position from JS are ignored. Focus moves into the sheet after the 250ms slide.
JS: open / position / close
JS
/* ── Pack Size Picker ──────────────────────────────────────
   One instance per page, filled per product. Edits the cart live:
   every stepper change is applied immediately (no submit button). */
var picker = { id: null, trigger: null };

function openPicker(id, trigger) {
  picker.id = id; picker.trigger = trigger;
  var el = document.getElementById('pack-picker');
  document.getElementById('pack-picker-title').textContent = PRODUCTS[id].name;
  renderPicker(false);
  el.classList.add('is-open');
  document.getElementById('pack-picker-backdrop').classList.add('is-open');
  positionPicker(trigger);
  positionToasts();
  // Focus the dialog itself (no ring, nothing pre-selected); Tab moves to its first control.
  // On mobile wait for the sheet to finish sliding in.
  setTimeout(function () { el.focus({ preventScroll: true }); }, isMobile() ? 260 : 0);
}

function renderPicker(animate) {
  var p = PRODUCTS[picker.id], l = lineOf(picker.id);
  document.getElementById('pack-picker-rows').innerHTML = ['single', 'multi'].map(function (pack) {
    return packRowHTML(p, pack, { points: '', price: '<b>' + money(packPrice(p, pack)) + '</b>' },
      stepperHTML({ p: p, pack: pack, value: l[pack] || 0, scope: 'picker' }));
  }).join('');
  var units = lineUnits(p, l);
  document.getElementById('pack-picker-count').textContent =
    (units === 1 ? t('itemInCart') : t('itemsInCart', { n: units })) +
    (l.multi ? ' (' + (l.single || 0) + ' single + ' + l.multi + ' × ' + p.multi + ')' : '');
  document.getElementById('pack-picker-total').textContent = money(lineTotal(p, l));
  var pill = document.querySelector('#pack-picker-points .ufs-points');
  (animate ? animatePoints : setPoints)(pill, units * p.pts);
}

function pickerStep(pack, d) { pickerSet(pack, (lineOf(picker.id)[pack] || 0) + d, true); }
function pickerSet(pack, value, fromStep) {
  var before = setLine(picker.id, pack, value), q = lineOf(picker.id)[pack] || 0;   // setLine re-renders the picker
  if (q === before) { renderPicker(false); return; }                                 // e.g. "abc" typed: restore the field
  var row = document.querySelector('#pack-picker-rows [data-row="' + pack + '"]');
  rollQty(row.querySelector('.fi-qty'), before, q > before);
  if (q > before) flyToCart(row.querySelector('.ufs-pack-row__img'), IMG + PRODUCTS[picker.id].img);
  else bumpBadge();
  if (fromStep) {                                     // focus stays on the pressed button…
    var btn = row.querySelector(q < before ? '[data-role="dec"]' : '[data-role="inc"]');
    if (btn.disabled) btn = row.querySelector('[data-role="inc"]');   // …unless minus just disabled at 0
    btn.focus({ preventScroll: true });
  }
}

function closePicker(skipFocus) {
  if (!picker.id) return;
  document.getElementById('pack-picker').classList.remove('is-open');
  document.getElementById('pack-picker-backdrop').classList.remove('is-open');
  var back = pickerTrigger();
  if (!skipFocus && back) back.focus({ preventScroll: true });
  picker.id = null;
  positionToasts();
}

// While the mobile sheet is open, toasts sit 12px above it instead of behind it
function positionToasts() {
  var stack = document.getElementById('cart-toasts'), el = document.getElementById('pack-picker');
  if (picker.id && isMobile()) stack.style.bottom = 'calc(' + el.offsetHeight + 'px + var(--space-3))';
  else stack.style.bottom = '';
}

// The trigger may have been re-rendered while the picker was open (the tile behind it
// updates live), so fall back to the product's current Edit / Add to cart control.
function pickerTrigger() {
  if (picker.trigger && document.contains(picker.trigger)) return picker.trigger;
  return document.querySelector('[data-picker-return="' + picker.id + '"]');
}

function isMobile() { return window.matchMedia('(max-width: 768px)').matches; }

// Desktop: always ABOVE the trigger, 12px gap, same width as the tile and flush with
// its edges; the arrow (bottom edge) points at the trigger's centre. If there isn't
// room above, the page first smooth-scrolls just enough to fit it, 16px below the sticky
// header, with the picker riding along with the tile.
function pickerSafeTop() {
  var header = document.querySelector('[data-sticky-header]');   // the site header, while it's on screen
  return Math.max(0, header ? header.getBoundingClientRect().bottom : 0) + 16;
}
function positionPicker(trigger) {
  var el = document.getElementById('pack-picker'), arrow = el.querySelector('.ufs-pack-picker__arrow');
  if (isMobile()) { el.style.width = ''; return; }
  var tile = trigger.closest('[data-picker-anchor]'), box = tile && tile.getBoundingClientRect();
  if (box) el.style.width = box.width + 'px';
  var h = el.offsetHeight, need = h + 12 + pickerSafeTop(), r = trigger.getBoundingClientRect();
  if (r.top < need - 1 && scrollY > 0 && !picker.autoScroll) startPickerScroll(need - r.top);
  var w = el.offsetWidth;
  var left = box ? box.left : Math.min(Math.max(16, r.left + r.width / 2 - w / 2), innerWidth - w - 16);
  el.style.left = left + 'px';
  // While scrolling-to-fit it follows the tile; afterwards the floor only applies when
  // the page can't scroll further
  el.style.top = (picker.autoScroll ? r.top - 12 - h : Math.max(pickerSafeTop(), r.top - 12 - h)) + 'px';
  arrow.style.left = (r.left + r.width / 2 - left - 7) + 'px';
}

/* ── Cart state ──────────────────────────────────────────────
   One line per product, one quantity per pack type:
   cart = { tomato: { single: 2, multi: 1 } }
   Every change (tile, picker, hero, cart panel) goes through setLine(),
   and every view re-renders from this one state. Persist it however the
   platform stores the cart; the prototype keeps it in localStorage. */
var cart = {};
var cartViews = [];                       // render functions called after each change
var reduceMotion = window.matchMedia('(prefers-reduced-motion: reduce)');

function lineOf(id) { return cart[id] || { single: 0, multi: 0 }; }
function clampQty(v) { v = parseInt(v, 10); return isNaN(v) || v < 0 ? 0 : Math.min(v, 999); }
function packLabel(p, pack) { return pack === 'single' ? t('single') : t('packOf', { n: p.multi }); }
function packPrice(p, pack) { return pack === 'single' ? p.price : p.price * p.multi; }
function lineUnits(p, l) { return (l.single || 0) + (l.multi || 0) * (p.multi || 1); }
function lineTotal(p, l) { return (l.single || 0) * p.price + (p.multi ? (l.multi || 0) * packPrice(p, 'multi') : 0); }
function money(n) { return '€' + n.toFixed(2); }        // use Intl.NumberFormat(locale, { style: 'currency' }) in production
function fmtPoints(n) { return n.toLocaleString('de-DE'); }
function escHTML(s) { return String(s).replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/"/g, '&quot;'); }

// Returns the previous quantity, so callers can animate old → new
function setLine(id, pack, qty, quiet) {
  var l = lineOf(id), before = l[pack] || 0, q = clampQty(qty);
  if (q === before) return before;
  l = { single: l.single || 0, multi: l.multi || 0 };
  l[pack] = q;
  if (l.single + l.multi) cart[id] = l; else delete cart[id];
  renderCartBadge();
  if (!quiet) cartToast(PRODUCTS[id], pack, before, q);
  cartViews.forEach(function (render) { render(id); });
  return before;
}

function cartCount() {
  return Object.keys(cart).reduce(function (n, id) { return n + cart[id].single + cart[id].multi; }, 0);
}
function renderCartBadge() {
  var n = cartCount();
  document.querySelectorAll('[data-cart-button]').forEach(function (btn) {
    var badge = btn.querySelector('.ufs-nav__badge');
    badge.textContent = n; badge.hidden = n === 0;
    btn.setAttribute('aria-label', 'Cart, ' + n + ' item' + (n === 1 ? '' : 's'));
  });
}

/* ── Cart feedback ── */
// One wording for every change: Added / Updated / Removed
function cartToast(p, pack, before, q) {
  var what = p.name + (p.multi ? ' · ' + packLabel(p, pack) : '');
  var msg = !q ? t('removed', { what: what })
    : !before ? t('added', { what: what + (q > 1 ? ' × ' + q : '') })
    : t('updated', { what: what + ' × ' + q });
  showToast(msg, { label: t('viewCart'), fn: openCartPanel });
}

var TOAST_MAX = 2, TOAST_MS = 3200;
function showToast(msg, action) {
  var stack = document.getElementById('cart-toasts'), el = document.createElement('div');
  el.className = 'ufs-cart-toast';
  var text = document.createElement('span');
  text.textContent = msg;
  el.appendChild(text);
  if (action) {
    var b = document.createElement('button');
    b.type = 'button'; b.textContent = action.label;
    b.onclick = function () { action.fn(); dismissToast(el); };
    el.appendChild(b);
  }
  stack.appendChild(el);
  var live = [].filter.call(stack.children, function (c) { return !c.classList.contains('is-leaving'); });
  while (live.length > TOAST_MAX) dismissToast(live.shift());      // oldest leaves first
  if (!reduceMotion.matches) {                                      // grows in, so older toasts glide up
    var h = el.offsetHeight; el.style.overflow = 'hidden';
    el.animate([{ height: 0, opacity: 0, transform: 'translateY(8px)' }, { height: h + 'px', opacity: 1, transform: 'none' }],
      { duration: 220, easing: 'cubic-bezier(0.2, 0.8, 0.2, 1)' }).onfinish = function () { el.style.overflow = ''; };
  }
  el._timer = setTimeout(function () { dismissToast(el); }, TOAST_MS);
}
function dismissToast(el) {
  if (el.classList.contains('is-leaving')) return;
  el.classList.add('is-leaving'); clearTimeout(el._timer);
  if (reduceMotion.matches) { el.remove(); return; }
  var h = el.offsetHeight; el.style.overflow = 'hidden';
  el.animate([{ height: h + 'px', opacity: 1 }, { height: 0, opacity: 0 }],
    { duration: 200, easing: 'ease-in', fill: 'forwards' }).onfinish = function () { el.remove(); };
}

// Number roll over a stepper's input; the input's own text hides until it lands
function rollQty(qty, from, up) {

  var cs = getComputedStyle(input), roll = document.createElement('span');

function animatePoints(pill, to) {
  var val = pill.querySelector('.ufs-points__val');
  var from = parseInt(val.getAttribute('data-value') || '0', 10);
  val.setAttribute('data-value', to);

  pill.classList.remove('is-pulse'); void pill.offsetWidth; pill.classList.add('is-pulse');
  var start = performance.now();
  (function tick(now) {
    var k = Math.min(Math.max((now - start) / 600, 0), 1), e = 1 - Math.pow(1 - k, 3);   // ease-out cubic
    val.textContent = fmtPoints(Math.round(from + (to - from) * e));
    if (k < 1) pill._raf = requestAnimationFrame(tick);
  })(start);
}

Internationalisation

i18n & RTL

Every string goes through t(key) (see TEXT in the JS). Prices and quantities are wrapped in <bdi>, so "€18.90" and "× 6" keep their order inside Arabic text. The layout uses logical properties (inset-inline-end, justify-content), so rows, steppers and the summary mirror on their own. The steppers mirror too: minus sits on the right and plus on the left, following the reading direction.

ElementRule
