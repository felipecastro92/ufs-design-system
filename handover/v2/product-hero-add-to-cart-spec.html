Product Hero + Add to Cart

Webshop Product Hero: the CTA becomes a buy block with one row per pack size (Single item / Pack of N), each with its own Add to cart, plus a running summary with points.

Overview

Full Component Preview

The webshop version of the v1 Product Hero (product-hero-spec.html). The CTA (__cta / __ctas) is replaced by a buy block: one pack row per pack size, each with its own Add to cart, a running summary, and a sample link. Gallery, title, tags, USPs, SKU and utility bar are the v1 hero, unchanged. Unlike the tile, there is no popover: both pack sizes are always visible.

💡
Live: add either pack, step it, type a quantity, or step down to the bin. The summary, points, sticky bar copy, toasts and fly-to-cart all update.

Anatomy

The buy block takes the place of the v1 CTA in .ufs-product-hero__info, after the intro and before the SKU. The pack rows, cart stepper, points pill and summary are the shared components from the Pack Size Picker spec, so only the block wrapper and its spacing are new here.

Behavior

Add to Cart Flow

No picker and no confirm step: each row writes to the cart directly through the shared setLine(), and every change gets the same toast, fly-to-cart and badge feedback as the tile.

JS
Needs the shared cart engine and pack rows from pack-size-picker-spec.html.
JS
/* ── Product Hero + Add to cart ──
   Uses the shared cart engine (pack-size-picker-spec.html). */
TEXT.heroParts = '{parts}';
TEXT.partSingleHero = '{q} × single';
TEXT.partMultiHero = '{q} × pack of {n}';
TEXT.inCartSticky = 'In cart: {parts} · {total}';

var HERO_ID = 'granules';                 // the product on this page
function heroRoot() { return document.getElementById('hero-buy'); }

function heroParts(p, l) {
  var parts = [];
  if (l.single) parts.push(t('partSingleHero', { q: l.single }));
  if (l.multi) parts.push(t('partMultiHero', { q: l.multi, n: p.multi }));
  return parts.join(' · ');
}

// "€18.90 · 1 kg"  /  "€113.40 · 6 × 1 kg (€18.90 each)"
function heroMeta(p, pack) {
  return pack === 'single'
    ? '<b>' + money(p.price) + '</b> · ' + p.unit
    : '<b>' + money(packPrice(p, 'multi')) + '</b> · ' + p.multi + ' × ' + p.unit + ' (' + money(p.price) + ' ' + t('each') + ')';
}

// At 0 the row shows "Add to cart"; from 1 a stepper whose minus is a bin at 1
function heroControl(pack) {
  var p = PRODUCTS[HERO_ID], q = lineOf(HERO_ID)[pack] || 0;
  if (!q) {
    return '<button type="button" class="btn btn-primary ufs-product-hero__add" data-add="' + pack + '" onclick="heroStep(\'' + pack + '\',1)"' +
      ' aria-label="' + escHTML(t('add') + ': ' + packLabel(p, pack)) + '"><img src="../icons/cart.svg" alt=""/>' + t('add') + '</button>';
  }
  return stepperHTML({ p: p, pack: pack, value: q, scope: 'hero', deleteAtOne: true });
}

// Rows render once; afterwards only the controls and summary update, so focus survives
function renderHeroRows() {
  var p = PRODUCTS[HERO_ID], packs = p.multi ? ['single', 'multi'] : ['single'];
  heroRoot().querySelector('[data-packs]').innerHTML = packs.map(function (pack) {
    var pts = pointsHTML(p.pts * (pack === 'multi' ? p.multi : 1));     // fixed: points one pack earns
    return packRowHTML(p, pack, { points: pts, price: heroMeta(p, pack) }, heroControl(pack));
  }).join('');
  heroRoot().querySelector('[data-summary-points]').innerHTML = pointsHTML(0);
}

function renderHero(animate) {
  var root = heroRoot(), p = PRODUCTS[HERO_ID], l = lineOf(HERO_ID), units = lineUnits(p, l);
  root.querySelectorAll('[data-ctl]').forEach(function (ctl) { ctl.innerHTML = heroControl(ctl.getAttribute('data-ctl')); });
  root.querySelector('[data-summary]').hidden = units === 0;
  root.querySelector('[data-summary-count]').textContent =
    (units === 1 ? t('itemInCart') : t('itemsInCart', { n: units })) + (l.multi ? ' (' + heroParts(p, l) + ')' : '');
  root.querySelector('[data-summary-total]').textContent = money(lineTotal(p, l));
  (animate ? animatePoints : setPoints)(root.querySelector('[data-summary-points] .ufs-points'), units * p.pts);
  renderHeroSticky();
}
cartViews.push(function (id) { if (id === HERO_ID) renderHero(true); });   // cart panel edits update the hero too

function heroStep(pack, d) { heroSet(pack, (lineOf(HERO_ID)[pack] || 0) + d, true); }
function heroSet(pack, value, fromStep) {
  var row = heroRoot().querySelector('[data-row="' + pack + '"]');
  var before = setLine(HERO_ID, pack, value), q = lineOf(HERO_ID)[pack] || 0;   // setLine re-renders the controls
  if (q === before) { renderHero(false); return; }
  if (before && q) rollQty(row.querySelector('.fi-qty'), before, q > before);
  if (q > before) flyToCart(row.querySelector('.ufs-pack-row__img'), IMG + PRODUCTS[HERO_ID].img); else bumpBadge();
  if (fromStep) {          // focus stays on the pressed control, or on Add to cart once the line is gone
    var f = !q ? row.querySelector('[data-add]') : row.querySelector(q < before ? '[data-role="dec"]' : '[data-role="inc"]');
    f.focus({ preventScroll: true });
  }
}

/* Sticky Bottom CTA (v1): shows when the hero's buy block scrolls out of view */
function renderHeroSticky() {
  var sub = document.getElementById('product-sticky-cta-sub'), btn = document.getElementById('product-sticky-cta-btn');
  if (!sub) return;
  var p = PRODUCTS[HERO_ID], l = lineOf(HERO_ID);
  sub.textContent = lineUnits(p, l)
    ? t('inCartSticky', { parts: heroParts(p, l), total: money(lineTotal(p, l)) })
    : t('single') + ' ' + money(p.price) + (p.multi ? ' · ' + packLabel(p, 'multi') + ' ' + money(packPrice(p, 'multi')) : '');
  btn.textContent = lineUnits(p, l) ? t('viewCart') : t('add');
}
function heroStickyPrimary() {
  if (lineUnits(PRODUCTS[HERO_ID], lineOf(HERO_ID))) { openCartPanel(); return; }
  var root = heroRoot(), first = root.querySelector('[data-add]');
  root.scrollIntoView({ behavior: reduceMotion.matches ? 'auto' : 'smooth', block: 'center' });
  if (first) setTimeout(function () { first.focus({ preventScroll: true }); }, 450);
}

Internationalisation

i18n & RTL

The rows mirror on their own (grid + logical properties): the thumbnail moves to the right and the control to the left, and the ×N count moves to the thumbnail's bottom-left. The header label and price note swap sides. Prices are in <bdi> so "6 × 1 كغ (€18.90 للقطعة)" keeps its order.
