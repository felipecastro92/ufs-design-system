Cart Side Panel

Mini cart in the v1 side drawer: one row per pack line with stepper, price and bin, plus a footer with total, loyalty points and Proceed to Checkout.

Overview

Full Component Preview

The mini cart. It opens from the header cart icon (and from a toast's "View cart") in the v1 drawer shell: 420px, slides in from the right, or from the left in RTL. Each pack line is its own row with a stepper, price and bin. The footer holds the total, the loyalty points and Proceed to Checkout. Editing here goes through the same cart engine as the tiles, picker and hero (pack-size-picker-spec.html), so every surface stays in sync.

💡
Live: open the panel, step a row, type a quantity, or bin a row and use "Add again" in the toast. The cart starts with three lines so there is something to edit.

Structure

Anatomy

Header, scrolling list and a pinned footer inside .signin-panel. Only the content is new: the shell (backdrop, slide, width, mobile width, RTL side) is the v1 drawer shared with Sign-in, Filters and the Wholesaler Picker.

Rules
ControlBehaviour
− / +±1. Minus stops at 1: it never removes the line.
Quantity fieldTyped, applied on change, clamped to 1–999 (0 or empty restores 1)
BinRemoves the line: the row fades out, then a toast offers "Add again" (restores the same pack at its previous quantity)
Product nameLink to the product page. The panel stays open in the history of the page that opened it.
Behavior

Open, Edit & Close

The panel is a modal dialog. Opening it closes the Pack Size Picker if one is open. Panel edits are quiet: the panel already shows the change, so it raises no toast and no fly-to-cart. Only the header badge bumps.

InputResult
Header cart icon (desktop or mobile) / toast "View cart"Opens. Focus moves to the close button after 50ms. Body scroll is locked. Cart buttons get aria-expanded="true".
Close / backdrop / EscCloses. Focus returns to whatever opened it, or to the cart icon when it was a toast (the toast is gone by then).
TabTrapped inside the panel
Re-render after an editThe focused control is restored in the new row: same button, field or bin. If minus just disabled itself at 1, focus moves to plus.
Proceed to CheckoutLink to the checkout cart step (template-checkout-webshop.html#cart)
Cart changes elsewhereThe panel listens to the cart engine and re-renders even while closed, so it is always current when opened

JS
/* ── Cart Side Panel ──
   Uses the shared cart engine (pack-size-picker-spec.html): every edit here goes
   through setLine(), so tiles, the picker and the hero stay in sync. */
TEXT.cartTitle = 'My shopping cart ({n})';
TEXT.totalItems = 'Total {n} items'; TEXT.totalItem = 'Total 1 item';
TEXT.cartEmpty = 'Your cart is empty.'; TEXT.continueShopping = 'Continue shopping';
TEXT.removedLine = '{what} has been removed from your cart.'; TEXT.addAgain = 'Add again';
TEXT.removeFromCart = 'Remove {what} from cart';

// One row per pack line, in the order products were added
function cartLines() {
  var out = [];
  Object.keys(cart).forEach(function (id) {
    var p = PRODUCTS[id], l = cart[id];
    ['single', 'multi'].forEach(function (pack) {
      if (l[pack]) out.push({ p: p, pack: pack, q: l[pack], price: l[pack] * packPrice(p, pack), label: p.multi ? packLabel(p, pack) : '' });
    });
  });
  return out;
}

function cartRowHTML(r) {
  var key = r.p.id + '|' + r.pack, name = escHTML(r.p.name) + (r.label ? ', ' + r.label : '');
  var pts = r.p.pts * (r.pack === 'multi' ? r.p.multi : 1);
  return '<div class="ufs-cart-row" data-row="' + key + '">' +
    '<div class="ufs-cart-row__img" style="background-image:url(\'' + IMG + r.p.img + '\')" role="img" aria-label="' + escHTML(r.p.name) + '">' +
      (r.pack === 'multi' ? '<span class="ufs-pack-row__count" aria-hidden="true">×' + r.p.multi + '</span>' : '') + '</div>' +
    '<div class="ufs-cart-row__info">' +
      '<p class="ufs-cart-row__name"><a href="' + (r.p.url || '#') + '">' + escHTML(r.p.name) + '</a>' +
        '<span class="ufs-cart-row__pack">' + (r.label ? '<span>' + r.label + '</span>' : '') + pointsHTML(pts) + '</span></p>' +
      // Minus stops at 1: removing is the row's bin button
      stepperHTML({ p: r.p, pack: r.pack, value: r.q, scope: 'cart', args: "'" + r.p.id + "','" + r.pack + "'", min: 1 }) +
      '<div class="ufs-cart-row__price"><b>' + money(r.price) + '</b><span class="ufs-fine">' + t('priceNote') + '</span></div>' +
    '</div>' +
    '<button type="button" class="ufs-cart-row__remove" aria-label="' + escHTML(t('removeFromCart', { what: name })) + '"' +
      ' onclick="cartRemove(\'' + r.p.id + '\',\'' + r.pack + '\')">' + ICON.del + '</button>' +
  '</div>';
}

function renderCartPanel() {
  var lines = cartLines(), list = document.getElementById('cart-list'), n = cartCount();
  var total = lines.reduce(function (a, r) { return a + r.price; }, 0);
  var pts = lines.reduce(function (a, r) { return a + r.q * (r.pack === 'multi' ? r.p.multi : 1) * r.p.pts; }, 0);
  // Re-rendering replaces the rows: remember which control had focus and put it back
  var a = document.activeElement, row = a && list.contains(a) ? a.closest('[data-row]') : null;
  var focusKey = row && row.getAttribute('data-row'), focusRole = a && (a.getAttribute('data-role') || (a.classList.contains('ufs-cart-row__remove') ? 'remove' : 'qty'));
  document.getElementById('cart-title').textContent = t('cartTitle', { n: n });
  document.getElementById('cart-total-label').textContent = n === 1 ? t('totalItem') : t('totalItems', { n: n });
  document.getElementById('cart-total').textContent = money(total);
  animatePoints(document.querySelector('#cart-points .ufs-points'), pts);
  document.getElementById('cart-ft').hidden = !lines.length;
  list.innerHTML = lines.length ? lines.map(cartRowHTML).join('')
    : '<div class="ufs-cart__empty"><p>' + t('cartEmpty') + '</p><button type="button" class="btn btn-secondary" onclick="closeCartPanel()">' + t('continueShopping') + '</button></div>';
  if (focusKey) {
    var r2 = list.querySelector('[data-row="' + focusKey + '"]'), sel = focusRole === 'remove' ? '.ufs-cart-row__remove' : focusRole === 'qty' ? '.fi-qty-input' : '[data-role="' + focusRole + '"]';
    var target = r2 && (r2.querySelector(sel + ':not(:disabled)') || r2.querySelector('[data-role="inc"]'))   // minus just disabled at 1 → plus
      || list.querySelector('.ufs-cart-row__remove') || document.querySelector('#cart-panel .signin-close');
    target.focus({ preventScroll: true });
  }
}
cartViews.push(renderCartPanel);

// Panel edits are quiet: the panel itself shows the change, so no toast
function cartStep(id, pack, d) { cartSet(id, pack, (lineOf(id)[pack] || 0) + d); }
// Rows are found by data-row in any list (panel, checkout cart step), so the same handlers serve both
function cartRowsOf(id, pack) { return document.querySelectorAll('.ufs-cart-row[data-row="' + id + '|' + pack + '"]'); }
function cartSet(id, pack, value) {
  var q = Math.max(1, clampQty(value)), before = setLine(id, pack, q, true);   // typed 0 keeps 1: removing is the bin
  if (q === before) { cartViews.forEach(function (render) { render(id); }); return; }   // e.g. "abc" typed: restore the field
  cartRowsOf(id, pack).forEach(function (row) { rollQty(row.querySelector('.fi-qty'), before, q > before); });
  bumpBadge();
}
function cartRemove(id, pack) {
  var p = PRODUCTS[id], before = lineOf(id)[pack] || 0, rows = cartRowsOf(id, pack);
  // "Add again" puts the same pack back at its previous quantity
  showToast(t('removedLine', { what: p.name + (p.multi ? ', ' + packLabel(p, pack) : '') }),
    { label: t('addAgain'), fn: function () { setLine(id, pack, (lineOf(id)[pack] || 0) + before, true); bumpBadge(); } });
  function done() { setLine(id, pack, 0, true); bumpBadge(); }
  if (!rows.length || reduceMotion.matches) { done(); return; }
  rows.forEach(function (row) { row.classList.add('is-removing'); });   // fade + slide 16px toward the inline-end
  setTimeout(done, 180);
}

var cartReturnFocus = null;
function cartIsOpen() { return document.getElementById('cart-panel').classList.contains('is-open'); }
function openCartPanel() {
  if (typeof closePicker === 'function') closePicker(true);
  var a = document.activeElement;   // opened from a toast's "View cart": return to the cart icon instead (the toast will be gone)
  cartReturnFocus = a && a.closest && a.closest('.ufs-cart-toast') ? document.querySelector('[data-cart-button]') : a;
  var panel = document.getElementById('cart-panel');
  panel.inert = false; panel.removeAttribute('aria-hidden');
  panel.classList.add('is-open'); document.getElementById('cart-backdrop').classList.add('is-open');
  document.body.style.overflow = 'hidden';
  document.querySelectorAll('[data-cart-button]').forEach(function (b) { b.setAttribute('aria-expanded', 'true'); });
  setTimeout(function () { panel.querySelector('.signin-close').focus(); }, 50);
}
function closeCartPanel() {
  if (!cartIsOpen()) return;
  var panel = document.getElementById('cart-panel');
  panel.classList.remove('is-open'); document.getElementById('cart-backdrop').classList.remove('is-open');
  panel.inert = true; panel.setAttribute('aria-hidden', 'true');
  document.body.style.overflow = '';
  document.querySelectorAll('[data-cart-button]').forEach(function (b) { b.setAttribute('aria-expanded', 'false'); });
  var back = cartReturnFocus && document.contains(cartReturnFocus) ? cartReturnFocus : document.querySelector('[data-cart-button]');
  back.focus({ preventScroll: true });
}
document.addEventListener('keydown', function (e) {
  if (!cartIsOpen()) return;
  if (e.key === 'Escape') { closeCartPanel(); return; }
  if (e.key !== 'Tab') return;                       // keep Tab inside the open panel
  var f = [].filter.call(document.querySelectorAll('#cart-panel button, #cart-panel input, #cart-panel a[href]'), function (el) { return el.offsetParent && !el.disabled; });
  if (e.shiftKey && document.activeElement === f[0]) { e.preventDefault(); f[f.length - 1].focus(); }
  else if (!e.shiftKey && document.activeElement === f[f.length - 1]) { e.preventDefault(); f[0].focus(); }
});
// The header cart buttons own the panel
document.querySelectorAll('[data-cart-button]').forEach(function (b) {
  b.setAttribute('aria-controls', 'cart-panel'); b.setAttribute('aria-haspopup', 'dialog'); b.setAttribute('aria-expanded', 'false');
});
